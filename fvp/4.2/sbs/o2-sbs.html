O2 passo a passo: Listagem e detalhe de Eventos para o Participante

Pedido típico: "o Participante precisa ver os Eventos" (/events e /events/{id}, só leitura, paginado, sem Novo/Editar/Remover, link na navbar). Chance: média-alta. Tempo: 25-35 min. Pré-requisito: nenhum (parte de main). É a base de P2 (inscrição).

Exemplo pronto: fvp/4.2/exemplos/o2-eventos-para-participante/ (patch, bundle e árvore projeto/). Este arquivo é o mesmo conteúdo em formato "um arquivo por vez, com o motivo".

Ordem = fluxo da requisição: rota -> controller -> views -> navbar -> testes. Sem banco, sem model (schema não muda). Na prova, prioridade: funcionar no navegador, depois testar, depois phpcs.

Por quê:

  • Grupo auth = qualquer logado (Participante e admin). Grupo admin aninhado barra quem não é admin. Rota do Participante tem que ficar só no auth.
  • {page} e {id} viram parâmetros do Request ($request->getParam('page')).

2. app/Controllers/EventsController.php (novo)

Cópia de index/show do app/Controllers/Admin/EventsController.php, namespace App\Controllers (não Admin), outra rota e outra view:

<?php

namespace App\Controllers;

use App\Models\Event;
use Core\Http\Controllers\Controller;
use Core\Http\Request;
use Lib\FlashMessage;

/**
 * Eventos na área do usuário (Participante): só listagem e detalhe, sem ações de administração.
 */
class EventsController extends Controller
{
    private const PER_PAGE = 5;

    public function index(Request $request): void
    {
        $paginator = Event::paginate(
            page: max(1, (int) $request->getParam('page', 1)),
            per_page: self::PER_PAGE,
            route: 'events.paginate'
        );
        $events = $paginator->registers();

        $title = 'Eventos';
        $this->render('events/index', compact('events', 'paginator', 'title'));
    }

    public function show(Request $request): void
    {
        $event = Event::findById((int) $request->getParam('id'));
        if ($event === null) {
            FlashMessage::danger('Evento não encontrado!');
            $this->redirectTo(route('events.index'));
            return;
        }

        $title = htmlspecialchars($event->title);
        $this->render('events/show', compact('event', 'title'));
    }
}

Por quê:

  • Controller separado: o middleware admin protege o controller do admin inteiro; misturar ações de Participante ali obrigaria a tirar a proteção ou duplicar rotas.
  • Namespace App\Controllers + arquivo em app/Controllers/ = PSR-4. Classe nova -> dump-autoload.
  • route: 'events.paginate': o _pages.phtml monta os links com $paginator->getRouteName(). Copiar o route: 'admin.events.paginate' do admin sem trocar: os links da paginação vão para /admin/events/page/N e o Participante cai em /dashboard. (Sem route: nenhum, o padrão "$table.paginate" já daria events.paginate.)
  • $title no compact: o layout imprime <?= $title ?> no <h1>. Esquecer -> 500.
  • return depois do redirectTo: em produção ele já dá exit, mas no ControllerTestCase o redirectTo só imprime Location: ...; sem o return, o teste segue e renderiza com null.
  • Controller fino: só getParam, FlashMessage, redirectTo(route()) e render.

namespace Tests\Integration\Access;

use App\Models\Event;
use App\Models\User;
use GuzzleHttp\Client;
use PHPUnit\Framework\Attributes\DataProvider;
use Tests\TestCase;

/**
 * Rotas events.* (área do usuário): qualquer usuário autenticado acessa; sem sessão vai para /login.
 */
class ParticipantEventsAccessTest extends TestCase
{
    private const PASSWORD = '123456';

    private User $admin;
    private User $participant;
    private Event $event;

    public function setUp(): void
    {
        parent::setUp();

        $this->admin = $this->createUser('Administrador', 'admin@example.com', 'admin');
        $this->participant = $this->createUser('Participante', 'participante@example.com', 'participant');

        $this->event = new Event([

            'title' => 'Palestra de Acesso',
            'description' => 'Evento usado nos testes de acesso.',
            'kind' => 'lecture',
            'modality' => 'online',
            'audience' => 'students',
            'starts_on' => date('Y-m-d', strtotime('+30 days')),
            'location' => 'Microsoft Teams',
            'capacity' => 10,
        ]);
        $this->event->save();
    }

    private function createUser(string $name, string $email, string $role): User
    {
        $user = new User([
            'name' => $name,
            'email' => $email,
            'password' => self::PASSWORD,
            'password_confirmation' => self::PASSWORD,
            'role' => $role,
        ]);
        $user->save();

        return $user;
    }

    private function newClient(): Client
    {
        return new Client([
            'allow_redirects' => false,
            'base_uri' => 'http://web:8080',
            'cookies' => true,
        ]);
    }

    private function loggedClient(User $user): Client
    {
        $client = $this->newClient();
        $response = $client->post('/login', [
            'form_params' => ['user' => ['email' => $user->email, 'password' => self::PASSWORD]],
        ]);

        $this->assertEquals(302, $response->getStatusCode());

        return $client;
    }

    /**
     * @return array<string, array{string}>
     */
    public static function participantRoutes(): array
    {
        return [
            'events.index' => ['/events'],
            'events.paginate' => ['/events/page/1'],
            'events.show' => ['/events/1'],
        ];
    }

    #[DataProvider('participantRoutes')]
    public function test_events_route_should_redirect_to_login_without_session(string $uri): void
    {
        $response = $this->newClient()->get($uri);

        $this->assertEquals(302, $response->getStatusCode());
        $this->assertEquals('/login', $response->getHeaderLine('Location'));
    }

    #[DataProvider('participantRoutes')]
    public function test_participant_should_access_events_route(string $uri): void
    {
        $response = $this->loggedClient($this->participant)->get($uri);

        $this->assertEquals(200, $response->getStatusCode());
        $this->assertStringContainsString('Palestra de Acesso', (string) $response->getBody());

    public function test_admin_should_also_access_participant_events(): void

        $this->assertEquals(200, $response->getStatusCode());
        $this->assertStringContainsString('Palestra de Acesso', (string) $response->getBody());
